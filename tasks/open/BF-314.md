import subprocess
import sys
from collections import defaultdict

base, head = sys.argv[1:3]
OURS = "75296019+EagleBearerYS@users.noreply.github.com"


def git(*args):
    return subprocess.run(["git", *args], capture_output=True, text=True, encoding="utf-8", errors="replace").stdout


merges = git("log", "--first-parent", "--merges", "--format=%H %s", f"{base}..{head}").splitlines()
for line in merges:
    sha, subject = line.split(" ", 1)
    parent = sha + "^1"
    diff = git("diff", "-U0", parent, sha)
    removed = defaultdict(list)
    current = None
    for row in diff.splitlines():
        if row.startswith("--- "):
            current = row[6:] if row.startswith("--- a/") else None
        elif row.startswith("@@") and current:
            match = re.match(r"@@ -(\d+)(?:,(\d+))? \+", row)
            start = int(match.group(1))
            count = int(match.group(2)) if match.group(2) is not None else 1
            if count > 0:
                removed[current].append((start, start + count - 1))
    hits = {}
    for path, ranges in removed.items():
        if path.startswith("docs/") or "/tests/" in path or path.endswith(".md"):
            continue
        args = ["blame", "--line-porcelain", parent]
        for first, last in ranges:
            args += ["-L", f"{first},{last}"]
        args += ["--", path]
        out = git(*args)
        commits = defaultdict(int)
        sha_line = None
        for row in out.splitlines():
            if re.match(r"^[0-9a-f]{40} ", row):
                sha_line = row.split()[0]
            elif row.startswith("author-mail ") and OURS in row:
                commits[sha_line[:10]] += 1
        if commits:
            hits[path] = dict(commits)
    if hits:
        number = re.search(r"#(\d+)", subject)
        print(f"{sha[:10]} {('#' + number.group(1)) if number else ''} {subject[:90]}")
        for path, commits in hits.items():
            total = sum(commits.values())
            print(f"    {path}: {total} of our lines replaced or removed, from {', '.join(f'{c} ({n})' for c, n in commits.items())}")
