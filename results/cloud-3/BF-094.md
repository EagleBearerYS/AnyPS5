## 3. #6921 on this branch: the `PageSpan` a rebase would take

This is the resolution used on the local merge. #6921's other hunks (`BlockPages`, `record()` calling `store(first, last, value)`, `forgetInaccessible()`, the tests) merge without change.

```cpp
struct PageSpan {
    std::uintptr_t base = 0;
    std::size_t size = 0;
    // Per page: access flags, zero while unknown.
    TouchedTable<std::uint8_t> pages;
    TouchedTable<std::uint8_t> blocks;

    bool allocate() {
        if (size == 0) return false;
#ifdef _WIN32
        constexpr auto backing = TableBacking::Reserved;
#else
        constexpr auto backing = TableBacking::Heap;
#endif
        const auto count = (size / PageBytes + BlockPages) & ~(BlockPages - 1);
        if (!pages.Allocate(count, backing) || !blocks.Allocate((count / BlockPages + 8) & ~static_cast<std::size_t>(7), backing)) size = 0;
        return size != 0;
    }

    bool covers(std::uintptr_t address) const {
        return size != 0 && address >= base && address - base < size;
    }

    std::uint8_t load(std::uintptr_t address) const {
        return pages.Load((address - base) / PageBytes);
    }

    std::uint8_t common(std::size_t block) const {
        const auto first = block * BlockPages;
        const auto value = pages.Load(first);
        return pages.RunEnd(first, value, first + BlockPages) == first + BlockPages ? value : 0;
    }

    void settle(std::size_t first, std::size_t last) {
        std::atomic_thread_fence(std::memory_order_seq_cst);
        for (auto block = first / BlockPages; block <= (last - 1) / BlockPages; ++block) {
            const auto value = common(block);
            if (blocks.Load(block) == value) continue;
            blocks.Store(block, value);
            if (value == 0) continue;
            std::atomic_thread_fence(std::memory_order_seq_cst);
            if (common(block) != value) blocks.Store(block, 0);
        }
    }

    void store(std::uintptr_t first, std::uintptr_t last, std::uint8_t value) {
        if (first >= last) return;
        const auto begin = (first - base) / PageBytes;
        const auto end = (last - 1 - base) / PageBytes + 1;
        for (auto index = begin; index < end; ++index) pages.Store(index, value);
        settle(begin, end);
    }

    std::uintptr_t runEnd(std::uintptr_t address, std::uint8_t value, std::uintptr_t limit) const {
        auto index = (address - base) / PageBytes;
        const auto stop = std::min<std::uintptr_t>((std::min(limit, base + size) - base + PageBytes - 1) / PageBytes, size / PageBytes);
        while (index < stop) {
            if (index % BlockPages == 0) {
                const auto block = index / BlockPages;
                if (block % 8 == 0 && index + 8 * BlockPages <= stop && blocks.RunEnd(block, value, block + 8) == block + 8) {
                    index += 8 * BlockPages;
                    continue;
                }
                if (index + BlockPages <= stop && blocks.Load(block) == value) {
                    index += BlockPages;
                    continue;
                }
            }
            const auto next = std::min<std::uintptr_t>(stop, (index / BlockPages + 1) * BlockPages);
            index = pages.RunEnd(index, value, next);
            if (index < next) break;
        }
        return std::min(limit, base + index * PageBytes);
    }

    void forget(std::uintptr_t address, std::size_t bytes) {
        if (size == 0 || bytes == 0 || address >= base + size || address + bytes <= base) return;
        const auto first = (std::max(address, base) - base) / PageBytes;
        const auto last = (std::min(address + bytes, base + size) - 1 - base) / PageBytes;
        pages.Clear(first, last + 1);
        std::atomic_thread_fence(std::memory_order_seq_cst);
        for (auto block = first / BlockPages; block <= last / BlockPages; ++block) {
            if (blocks.Load(block) != 0) blocks.Store(block, 0);
        }
    }
};
```

- **The block check:** `common()` asks "are all 64 pages equal to the first" through `RunEnd`, which does #6921's 8-byte loads.
  - A block lies inside one 64 KiB table page (65536 is a multiple of 64), so a block is either all untouched (reads 0, `common` = 0) or all committed.
- **A block entry of 0** means "not uniform" in #6921, as before.
  - #6921's only `runEnd` caller (`describePages`) passes a nonzero value, so the block-level `== value` skip never sees 0.
- **Reserved backing, worst case:** `blocks` costs one 64 KiB table page per 16 GiB of guest span touched, plus the page table's own pages.
