# PRM-101: an SGPR-base global load outside the BDA table crashes lavapipe on main too

**Answer:** this is main's behaviour, not something #8963 or #13858 brings in. All three trees crash in the same place, in the same way. Each one reports the `off`-form load's fault through the fault buffer as expected. There is no bisect, because main itself crashes.

- On every tree, the faulting instruction is in llvmpipe's JIT fragment code. There are no symbols and no AnyPS5 frame.
- It is a broadcast of one dword from a null pointer (`rax = 0`). This is what a wave-uniform load compiles to, so the SGPR-base load is emitted as a single scalar load.
- The `off` form's per-lane load doesn't crash. Its out-of-table lane reaches the fault buffer instead.
- **A reading, not checked against the recompiler's SPIR-V:** when the BDA lookup misses on the uniform path, the translated pointer is null, and the load from it is still executed.
- The 13858 gdb run didn't print rax. The command file only gained `rax` for the later runs.

None. Main (cd087bd12c) crashes, so neither #8963 nor #13858 causes it.

using AgcDriver::Graphics::ColorTargetLayout;

constexpr std::uint32_t Width = 64;
constexpr std::uint32_t Height = 64;
constexpr std::size_t BlockBytes = 65536;
constexpr std::size_t VertexCodeOffset = 0;
constexpr std::size_t PixelCodeOffset = 256;
constexpr std::size_t IndirectCodeOffset = 512;
constexpr std::size_t PointerSlot = 1024;

constexpr std::array<std::uint32_t, 6> VertexCode{
    0xe0382000u, 0x80000005u, 0xbf8c3f70u, 0xf80008cfu, 0x03020100u, 0xbf810000u,
};

constexpr std::array<std::uint32_t, 7> PixelCode{
    0x7e000280u,
    0xdc308000u, 0x01000000u,
    0xbf8c3f70u,
    0xf800180fu, 0x01010101u,
    0xbf810000u,
};

constexpr std::array<std::uint32_t, 10> IndirectPixelCode{
    0x7e000280u,
    0xdc348000u, 0x02000000u,
    0xbf8c3f70u,
    0xdc308000u, 0x017d0002u,
    0xbf8c3f70u,
    0xf800180fu, 0x01010101u,

constexpr std::array<std::array<float, 4>, 3> Triangle{{
    {-1.0f, -1.0f, 0.5f, 1.0f}, {3.0f, -1.0f, 0.5f, 1.0f}, {-1.0f, 3.0f, 0.5f, 1.0f}
}};

const ColorTargetLayout TargetLayout(Width, Height, ColorTileMode::RenderTarget, 4);

std::uint64_t AddressOf(const void* data) {
    return static_cast<std::uint64_t>(reinterpret_cast<std::uintptr_t>(data));
}

class Pages {
public:
    explicit Pages(std::size_t bytes) : bytes(bytes) {
        if (watched) {
            void* mapped = mmap(nullptr, bytes + BlockBytes, PROT_READ | PROT_WRITE, MAP_PRIVATE | MAP_ANONYMOUS, -1, 0);
            Require(mapped != MAP_FAILED, "probe: cannot map guest pages");
            const auto begin = reinterpret_cast<std::uintptr_t>(mapped);
            const auto aligned = (begin + BlockBytes - 1) & ~(static_cast<std::uintptr_t>(BlockBytes) - 1);
            if (aligned != begin) munmap(mapped, aligned - begin);
            if (aligned + bytes != begin + bytes + BlockBytes) munmap(reinterpret_cast<void*>(aligned + bytes), begin + BlockBytes - aligned);
            data = reinterpret_cast<std::uint8_t*>(aligned);
            GuestWriteWatch::GuestWriteWatchRegister_nid_postfix(data, bytes);
        } else {
            data = static_cast<std::uint8_t*>(std::aligned_alloc(BlockBytes, bytes));
        }
        Require(data != nullptr, "probe: cannot allocate guest pages");
        std::memset(data, 0, bytes);
    }

    ~Pages() {
        AgcDriver::Graphics::StorageTexture::FlushPending(Address(), bytes, nullptr, "test release");
        if (registered) GuestAllocations::Mutation().Remove(data);
        if (watched) {
            munmap(data, bytes);
            GuestWriteWatch::GuestWriteWatchUnregister_nid_postfix(data, bytes);
        } else {
            std::free(data);
        }
    }

    Pages(const Pages&) = delete;
    Pages& operator=(const Pages&) = delete;

    void Register(bool writable) {
        GuestAllocations::Mutation().Add(data, bytes, true, writable, true);
        registered = true;
    }

    std::uint64_t Address(std::size_t offset = 0) const { return AddressOf(data + offset); }
    std::uint8_t* Data() { return data; }

private:
    std::size_t bytes;
    bool watched = AgcDriver::GuestMemory::WriteWatched();
    std::uint8_t* data = nullptr;
    bool registered = false;
};

class StderrCapture {
public:
    StderrCapture() {
        std::fflush(stderr);
        file = std::tmpfile();
        Require(file != nullptr, "cannot open a temporary file for stderr");
        saved = dup(fileno(stderr));
        dup2(fileno(file), fileno(stderr));
    }

    std::string Finish() {
        std::fflush(stderr);
        dup2(saved, fileno(stderr));
        close(saved);
        std::rewind(file);
        std::string text;
        char buffer[4096];
        for (std::size_t read = 0; (read = std::fread(buffer, 1, sizeof(buffer), file)) != 0;) text.append(buffer, read);
        std::fclose(file);
        std::fwrite(text.data(), 1, text.size(), stderr);
        return text;
    }

private:
    std::FILE* file = nullptr;
    int saved = -1;
};

class Unmapped {
public:
    Unmapped() {
        data = mmap(nullptr, BlockBytes, PROT_NONE, MAP_PRIVATE | MAP_ANONYMOUS, -1, 0);
        if (data == MAP_FAILED) data = nullptr;
        Require(data != nullptr, "probe: cannot reserve an inaccessible range");
    }

    ~Unmapped() { munmap(data, BlockBytes); }

    Unmapped(const Unmapped&) = delete;
    Unmapped& operator=(const Unmapped&) = delete;

    std::uint64_t Address(std::size_t offset) const { return AddressOf(static_cast<const std::uint8_t*>(data) + offset); }

private:
    void* data = nullptr;
};

struct Scene {
    AgcDriver::VulkanDevice& device;
    Pages pixels{TargetLayout.Bytes()};
    Pages vertices{BlockBytes};
    Pages code{BlockBytes};
    Pages constants{BlockBytes};
    std::vector<ShaderRecompiler::RecompileResult> vertexResults;
    std::uint32_t vertexPush = 0;
    std::vector<ShaderRecompiler::RecompileResult> pixelResults;
    std::array<AgcDriver::Graphics::CompiledShader, 2> shaders{};
    std::vector<AgcDriver::Graphics::GuestMemorySnapshot> snapshots;
    AgcDriver::Graphics::State state{};

    explicit Scene(AgcDriver::VulkanDevice& device) : device(device) {
        vertexResults.reserve(4);
        pixelResults.reserve(8);
        pixels.Register(true);
        std::memcpy(vertices.Data(), Triangle.data(), sizeof(Triangle));
        vertices.Register(false);
        std::memcpy(code.Data() + VertexCodeOffset, VertexCode.data(), sizeof(VertexCode));
        std::memcpy(code.Data() + PixelCodeOffset, PixelCode.data(), sizeof(PixelCode));
        std::memcpy(code.Data() + IndirectCodeOffset, IndirectPixelCode.data(), sizeof(IndirectPixelCode));
        code.Register(false);
        constants.Register(true);
        Fetch();
        state.stages = {AgcDriver::Graphics::ShaderPath::Vertex, 0u, 64u, 64u, std::nullopt, std::nullopt};
        state.color = {pixels.Address(), {Width, Height}, VK_FORMAT_R8G8B8A8_UNORM, TargetLayout.Bytes(), 0xe4u, ColorTileMode::RenderTarget};
        state.colors = {state.color};
        state.hasColorTarget = true;
        state.renderExtent = {Width, Height};
        state.topology = VK_PRIMITIVE_TOPOLOGY_TRIANGLE_LIST;
        state.viewport = {0, static_cast<float>(Height), static_cast<float>(Width), -static_cast<float>(Height), 0, 1};
        state.negativeOneToOne = false;
        state.scissor = {{0, 0}, {Width, Height}};
        state.cullMode = VK_CULL_MODE_NONE;
        state.frontFace = VK_FRONT_FACE_COUNTER_CLOCKWISE;
        state.blend.colorWriteMask = 15;
        state.blends = {state.blend};
        state.blendConstants = {};
    }

    void Fetch() {
        const auto address = vertices.Address();
        const std::vector<std::uint32_t> userData{static_cast<std::uint32_t>(address), static_cast<std::uint32_t>((address >> 32u) & 0xffffu) | (16u << 16u), static_cast<std::uint32_t>(BlockBytes / 16u), 0x01016facu};
        const std::array<ShaderRecompiler::MemoryRegion, 1> vertexMemory{{{code.Address(VertexCodeOffset), std::as_bytes(std::span(VertexCode))}}};
        ShaderRecompiler::RecompileRequest vertex{
            {ShaderStage::Vertex, code.Address(VertexCodeOffset), VertexCode, 0, {}},
            {64, 0, userData, std::nullopt, std::nullopt, ShaderRecompiler::ShaderVertexStageInfo{}, vertexMemory},
            device.Target(),
            {0, 0, 0, 64}
        };
        vertex.useCache = false;
        vertexResults.push_back(ShaderRecompiler::Recompile(vertex));
        vertexPush = static_cast<std::uint32_t>(vertexResults.back().pushConstants.size());
        shaders[0] = {ShaderStage::Vertex, &vertexResults.back(), 0};
    }

    void Point(std::uint64_t slot, std::span<const std::uint32_t> program, std::size_t offset) {
        const auto target = device.Target();
        ShaderRecompiler::ShaderPixelStageInfo pixel{};
        pixel.inputAddr = ShaderRecompiler::PixelInputBit(ShaderRecompiler::PixelInput::PositionX) | ShaderRecompiler::PixelInputBit(ShaderRecompiler::PixelInput::PositionY);
        pixel.posX = true;
        pixel.posY = true;
        pixel.targetOutputMode[0] = 9;
        pixel.targetExportMapping.fill(0xe4u);
        const std::vector<std::uint32_t> userData{static_cast<std::uint32_t>(slot), static_cast<std::uint32_t>(slot >> 32u), 0u, 0u};
        const std::array<ShaderRecompiler::MemoryRegion, 1> pixelMemory{{{code.Address(offset), std::as_bytes(program)}}};
        ShaderRecompiler::RecompileRequest fragment{
            {ShaderStage::Fragment, code.Address(offset), program, 0, {}},
            {64, 0, userData, std::nullopt, pixel, std::nullopt, pixelMemory},
            target,
            PixelPushLayout(vertexPush, target)
        };
        fragment.useCache = false;
        pixelResults.push_back(ShaderRecompiler::Recompile(fragment));
        const auto& pixelResult = pixelResults.back();
        Require(std::any_of(pixelResult.bindings.begin(), pixelResult.bindings.end(), [](const auto& binding) { return binding.role == ShaderRecompiler::DescriptorRole::BdaPagetable; }), "probe: the pixel shader does not read guest memory through the BDA table");
        shaders[1] = {ShaderStage::Fragment, &pixelResult, PixelPushOffset(vertexPush, target)};
        snapshots = {
            {code.Address(VertexCodeOffset), std::as_bytes(std::span(reinterpret_cast<const std::uint32_t*>(code.Data() + VertexCodeOffset), VertexCode.size()))},
            {code.Address(offset), std::as_bytes(std::span(reinterpret_cast<const std::uint32_t*>(code.Data() + offset), program.size()))},
        };
    }

    void Draw() {
        std::lock_guard gpu(AgcDriver::GuestMemory::GpuMutex());
        const AgcDriver::Pm4::DrawParameters draw{0, static_cast<std::uint32_t>(Triangle.size()), 0, 1, 0, false};
        device.Draw(state, draw, shaders, snapshots, nullptr);
    }

    void Wait() {
        std::lock_guard gpu(AgcDriver::GuestMemory::GpuMutex());
        device.WaitIdle();
    }

    std::uint8_t Color() {
        AgcDriver::Graphics::StorageTexture::FlushPending(pixels.Address(), TargetLayout.Bytes(), nullptr, "test read-back");
        Wait();
        std::vector<std::uint8_t> texels(TargetLayout.Bytes());
        AgcDriver::GuestMemory::Read(pixels.Address(), std::as_writable_bytes(std::span(texels)), 1);
        return texels[TargetLayout.Offset(0, 0)];
    }
};

std::size_t Occurrences(const std::string& text, const std::string& what) {
    std::size_t count = 0;
    for (auto at = text.find(what); at != std::string::npos; at = text.find(what, at + what.size())) ++count;
    return count;
}

std::string FaultLines(const std::string& text) {
    std::string lines;
    std::size_t at = 0;
    while (at < text.size()) {
        const auto end = text.find('\n', at);
        const auto line = text.substr(at, end == std::string::npos ? std::string::npos : end - at);
        if (line.find("BDA access failed") != std::string::npos) lines += "  " + line + "\n";
        if (end == std::string::npos) break;
        at = end + 1;
    }
    return lines;
}

void Report(const char* name, std::uint64_t address, const std::string& text, std::uint8_t color) {
    char where[64];
    std::snprintf(where, sizeof(where), "address=0x%llx ", static_cast<unsigned long long>(address));
    std::printf("probe %s: draw returned, color %u, %zu BDA faults reported, %zu at 0x%llx\n%s", name, static_cast<unsigned>(color), Occurrences(text, "BDA access failed"), Occurrences(text, where), static_cast<unsigned long long>(address), FaultLines(text).c_str());
    std::fflush(stdout);
}

void Good(Scene& scene) {
    const float level = 1.0f;
    std::memcpy(scene.constants.Data(), &level, sizeof(level));
    scene.Point(scene.constants.Address(), PixelCode, PixelCodeOffset);
    StderrCapture log;
    scene.Draw();
    const auto color = scene.Color();
    Report("good (sgpr base, mapped)", scene.constants.Address(), log.Finish(), color);
}

void Off(Scene& scene) {
    const Unmapped unmapped;
    const auto address = unmapped.Address(0x100);
    std::memcpy(scene.constants.Data() + PointerSlot, &address, sizeof(address));
    scene.Point(scene.constants.Address(PointerSlot), IndirectPixelCode, IndirectCodeOffset);
    std::printf("probe off: drawing global_load_dword v1, v[2:3], off at 0x%llx\n", static_cast<unsigned long long>(address));
    std::fflush(stdout);
    StderrCapture log;
    scene.Draw();
    const auto color = scene.Color();
    Report("off (vgpr address, unmapped)", address, log.Finish(), color);
}

void Sgpr(Scene& scene) {
    const Unmapped unmapped;
    const auto address = unmapped.Address(0x100);
    scene.Point(address, PixelCode, PixelCodeOffset);
    std::printf("probe sgpr: drawing global_load_dword v1, v0, s[0:1] at 0x%llx\n", static_cast<unsigned long long>(address));
    std::fflush(stdout);
    StderrCapture log;
    scene.Draw();
    const auto color = scene.Color();
    Report("sgpr (sgpr base, unmapped)", address, log.Finish(), color);
}

}

int main(int argc, char** argv) {
    try {
        const std::string mode = argc > 1 ? argv[1] : "";
        Require(mode == "good" || mode == "off" || mode == "sgpr", "usage: good | off | sgpr");
        const auto device = OpenVulkanTestDevice();
        if (!device) return VulkanTestSkipped;
        {
            Scene scene(*device);
            if (mode == "good") Good(scene);
            if (mode == "off") Off(scene);
            if (mode == "sgpr") Sgpr(scene);
            scene.Wait();
        }
        std::printf("probe %s finished\n", mode.c_str());
        return 0;
    } catch (const std::exception& error) {
        std::cerr << error.what() << '\n';
        return 1;
    }
}

## Follow-up, not done here

Why the uniform BDA load path dereferences a null pointer on a miss on lavapipe would be a task of its own. It would mean dumping the probe's SPIR-V for `sgpr` and `off` and comparing how a lookup miss is guarded on each. A hardware driver may behave differently (fault, or read zero). This was only run on lavapipe.
